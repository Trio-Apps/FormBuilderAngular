CREATE OR ALTER PROCEDURE [dbo].[GetFieldTypes]
    @IsActive BIT = 1,
    @HasOptions BIT = NULL,
    @DataType NVARCHAR(50) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        Id,
        TypeName,
        DataType,
        MaxLength,
        HasOptions,
        AllowMultiple,
        IsActive,
        CreatedDate,
        UpdatedDate
    FROM 
        [dbo].[FIELD_TYPES]
    WHERE 
        IsDeleted = 0
        AND (@IsActive IS NULL OR IsActive = @IsActive)
        AND (@HasOptions IS NULL OR HasOptions = @HasOptions)
        AND (@DataType IS NULL OR DataType = @DataType)
    ORDER BY 
        TypeName ASC;
END
